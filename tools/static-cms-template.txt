function renderPublicBody(entry: {body: string; renderedHtml?: string}) { return typeof entry.renderedHtml === 'string' ? entry.renderedHtml : markdownToHtml(entry.body); }
const entries: CmsEntry[] = publicData.entries.map(entry => ({
  ...entry, type: entry.type as EditorialType, language: entry.language as 'en' | 'zh', status: 'published',
  html: renderPublicBody(entry),
}));
const coreRoutes: Record<string, string> = { home: '/', about: '/about', bstvc: '/bstvc', research: '/research', publications: '/publications', resources: '/resources', news: '/news', journal: '/blogs' };
function route(entry: Pick<CmsEntry, 'type' | 'slug'>) {
  if (entry.type === 'page') return Object.hasOwn(coreRoutes, entry.slug) ? coreRoutes[entry.slug] : `/${encodeURIComponent(entry.slug)}`;
  const prefix = { news: 'news', publication: 'publications', journal: 'blogs', research: 'research', resource: 'resources' }[entry.type];
  return `/${prefix}/${encodeURIComponent(entry.slug)}`;
}

export type HomepageUpdate = { type: EditorialType; slug: string; title: string; date: string | null; updated: string | null; section: string; href: string };
export async function getHomepageUpdates(limit = 6): Promise<HomepageUpdate[]> {
  const sections: Record<string, string> = { news: 'News', publication: 'Publications', journal: 'Blogs', research: 'Research', resource: 'Resources' };
  const updates: HomepageUpdate[] = entries.filter(entry => entry.type !== 'page').map(entry => ({
    type: entry.type, slug: entry.slug, title: entry.title, date: entry.date, updated: entry.updated, section: sections[entry.type], href: route(entry),
  }));
  updates.push(...migrated.posts.map(post => ({ type: 'journal' as const, slug: post.slug, title: post.title, date: post.date, updated: post.modified ?? post.date, section: 'Blogs', href: `/blogs/${encodeURIComponent(post.slug)}` })));
  return updates.sort((a, b) => (b.date ?? b.updated ?? '').localeCompare(a.date ?? a.updated ?? '') || a.title.localeCompare(b.title)).slice(0, Math.min(12, Math.max(1, limit)));
}

/** Decode one display layer, preserving authored Markdown code literals. */
function publicSummaryText(value: string): string {
  return value.split(/(```[\s\S]*?```|~~~[\s\S]*?~~~|`+[^`]*`+)/g)
    .map((part, index) => index % 2 ? part : decodeHtmlEntities(part))
    .join("");
}

/** Strip markup before decoding: encoded tag examples remain literal text. */
function publicHtmlText(value: string): string {
  return decodeHtmlEntities(value
    .replace(/<style\b[\s\S]*?<\/style>/gi, " ")
    .replace(/<script\b[\s\S]*?<\/script>/gi, " ")
    .replace(/<[^>]+>/g, " "))
    .replace(/\s+/g, " ").trim();
}

export async function buildDynamicSearchIndex(): Promise<SearchRecord[]> {
  return [
    ...entries.map(entry => ({ title: entry.title, summary: publicSummaryText(entry.summary), body: publicHtmlText(entry.html), href: route(entry), type: entry.type, language: entry.language, date: entry.date })),
    ...migrated.pages.map(page => ({ title: page.title, summary: publicSummaryText(page.headings.slice(0, 4).join(' · ')), body: publicHtmlText(page.contentHtml), href: `/archive/${encodeURIComponent(page.slug)}`, type: 'archive page', language: page.language as 'en' | 'zh', date: page.modified })),
    ...migrated.posts.map(post => ({ title: post.title, summary: publicSummaryText(post.headings.slice(0, 4).join(' · ')), body: publicHtmlText(post.contentHtml), href: `/blogs/${encodeURIComponent(post.slug)}`, type: 'journal archive', language: post.language as 'en' | 'zh', date: post.date })),
  ];
}

function getRouteMetadataInfo(pagePath, publicData, migrated = {pages: [],posts: []}) {
  const trim = value => value.replace(/\/+$/, '') || '/';
  const route = entry => {
    const core = {home:'/',about:'/about',bstvc:'/bstvc',research:'/research',publications:'/publications',resources:'/resources',news:'/news',journal:'/blogs'};
    const prefix = {news:'news',publication:'publications',journal:'blogs',research:'research',resource:'resources'};
    return entry.type === 'page' ? (Object.hasOwn(core,entry.slug)?core[entry.slug]:'/'+encodeURIComponent(entry.slug)) : '/'+prefix[entry.type]+'/'+encodeURIComponent(entry.slug);
  };
  const page = trim(pagePath.split('/').map(part=>{try{return encodeURIComponent(decodeURIComponent(part));}catch{return part;}}).join('/'));
  const name = publicData.settings?.siteName || 'Chao Song';
  const entry = publicData.entries.find(value => trim(route(value)) === page);
  if (entry) return {title:entry.seoTitle?.trim() || entry.title,description:entry.seoDescription?.trim() || entry.summary || entry.title,path:route(entry),type:entry.type==='page'?'website':'article'};
  const coreFallback = {
    '/':['Home',publicData.settings?.siteDescription || 'Research in health, geography and Bayesian spatiotemporal statistics.'],
    '/about':['About','Academic experience, research interests, professional links, and collaborators.'],
    '/bstvc':['BSTVC','Bayesian Spatiotemporally Varying Coefficients modeling.'],
    '/research':['Research','Research in health and medical geography and Bayesian spatiotemporal statistics.'],
    '/publications':['Publications','The peer-reviewed publication record of Chao Song.'],
    '/resources':['Resources','Datasets, software, documentation, and reproducible resources.'],
    '/news':['News','Latest research news and the public news archive.'],
    '/blogs':['Blogs','Research notes, science communication, and bilingual blog posts.'],
  };
  if (Object.hasOwn(coreFallback,page)) {const [title,description]=coreFallback[page];return {title:title+' | '+name,description,path:page,type:'website'};}
  if (page === '/search') return {title:'Search | '+name,description:'Search '+name+"'s research, publications, news, resources and bilingual archive.",path:'/search',type:'website',robots:{index:false,follow:true}};
  if (page === '/archive') return {title:'Source Archive | '+name,description:'Preserved public source pages in their original order and language.',path:'/archive',type:'website'};
  const legacyBlog = page.startsWith('/blogs/');
  if (!legacyBlog && !page.startsWith('/archive/')) return null;
  let slug;try{slug=decodeURIComponent(page.slice(legacyBlog?'/blogs/'.length:'/archive/'.length));}catch{return null;}
  const archived = (legacyBlog ? migrated.posts ?? [] : migrated.pages ?? []).find(value=>value.slug===slug);
  if (!archived) return null;
  const description = archived.headings?.filter(Boolean).slice(0,4).join(' · ') || archived.title+' — preserved article in its original language.';
  return {title:archived.title+(legacyBlog?' | '+name:' | Source Archive'),description,path:(legacyBlog?'/blogs/':'/archive/')+encodeURIComponent(archived.slug),type:'article'};
}

export async function getStaticEntryMetadata(type: EditorialType | 'search' | 'archive', slug: string) {
  const selectedPath = type === 'search' ? '/search' : type === 'archive' ? '/archive'+(slug?'/'+encodeURIComponent(slug):'') : route({type,slug});
  const info = getRouteMetadataInfo(selectedPath,publicData,migrated); if (!info) return null;
  const settings = await getSiteSettings();
  const {title,description} = info;
  const image = siteAssetUrl(settings.defaultOgAssetId, settings.defaultOgUrl);
  return {title:{absolute:title},description,...(info.robots?{robots:info.robots}:{}),alternates:{canonical:info.path},openGraph:{type:info.type,title,description,url:info.path,images:[image]},twitter:{card:'summary_large_image',title,description,images:[image]}};
}
